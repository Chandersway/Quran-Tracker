export function resolveLocale(value: unknown): Locale {
  return value === 'nl' || value === 'ar' ? value : 'en';
}

export function translate(source: string, locale: Locale): string {
  const row = copy[source];
  return row?.[locale === 'en' ? 0 : locale === 'nl' ? 1 : 2] ?? source;
}

export function translateTemplate(source: string, locale: Locale, values: Record<string, string | number>): string {
  return translate(source, locale).replace(/\{(\w+)\}/g, (_match, key: string) => String(values[key] ?? `{${key}}`));
}

export function useI18n() {
  const store = useOptionalAppStore();
  const locale = resolveLocale(store?.state.preferences.language);
  return { locale, t: (source: string) => translate(source, locale), template: (source: string, values: Record<string, string | number>) => translateTemplate(source, locale, values) };
}

export function hasAllLocaleEntries(): boolean {
  return Object.values(copy).every(row => row.length === 3 && row.every(value => typeof value === 'string' && value.length > 0));
}

export function hasArabicLocaleEntries(): boolean {
  return Object.values(copy).every(row => /[\u0600-\u06ff]/.test(row[2]));
}
